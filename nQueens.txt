-- n-queens
-- flags: -hof, -imm, -npm, -mpp

-- example predicate
--p(A) = True
--p(B) = False


-- auxiliary definitions
-- logical rules ("_" to prevent clash with INPLA built in names)
and_(True,y)  = y
and_(False,y) = False
not_(True)  = False
not_(False) = True

-- if/then/else
ite(True,t,e)  = t
ite(False,t,e) = e

-- list append
app([],ys) = ys
app((x:xs),ys) = (x:(app(xs,ys)))

-- all: returns true if predicate ^f applied to each member of list is true
all([],^f)     = True
all((x:xs),^f) = and_(^f(x),all(xs,^f))

-- zip: takes two lists and pairs them up element-by-element, stopping when either list runs out 
zip([],ys) = []
zip(*,[])  = []
zip((x:xs),(y:ys)) = (Pair(x, y)) : (zip(xs,ys))

-- filter : takes a list and a predicate and returns all elements of list for which predicate is true
filter([],^f)     = []
--filter(x:xs,^f) = filterA(^f(x),x,xs,^f)
--filterA(True,x,xs,^f)  = (x:(filter(xs,^f)))
--filterA(False,x,xs,^f) = filter(xs,^f)
filter((x:xs),^f) = ite(^f(x),(x:(filter(xs,^f))),(filter(xs,^f)))

-- concatMap
concatMap ([],^f)     = []
concatMap ((x:xs),^f) = app(^f(x),concatMap(xs,^f))

-- map
map([],^f)     = []
map((x:xs),^f) = (^f(x):map(xs,^f))

-- absolute difference
absDiff(Z,y)       = y
absDiff(S(x),Z)    = S(x)
absDiff(S(x),S(y)) = absDiff(x,y)

-- unary number equality
eq(Z,Z)      = True
eq(Z,S(x))   = False
eq(S(x),Z)   = False
eq(S(x),S(y))= eq(x,y)

-- check : check if queen safe. (c:cs) = existing queen positions, q = column of queen, d = row distance from new queen to first element of (c:cs) 
check([],q,d)     = True
check((c:cs),q,d) = and_(and_(not_(eq(c,q)),not_(eq(absDiff(c,q),d))),check(cs,d,S(d)))

-- cols
cols(Z)    = []
cols(S(n)) = app(cols(n),(S(n):[]))

-- extend : find all ways to extend current board
extend(Z,qs)    = []
extend(S(x),qs) = ite(check(qs,S(x),Z),(S(x):qs):(extend(qs,x)),extend(qs,x))

-- queens
queens(Z)    = []:[]
queens(S(x)) = concatMap(extend(S(x),queens(x)))
